# v0.8.0 Unknown-Track Language Detection Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Detect the real language of `und` audio + text-subtitle tracks — text via `langdetect` (at scan), audio via `faster-whisper` tiny language-ID (on-demand + a default-on pre-conversion hook) — and write results back so the v0.7.29/30 output-metadata injection tags converted files correctly.

**Architecture:** A new isolated `backend/language_detection.py` module owns both detectors, confidence gating, and the ffmpeg clip-extraction. The scanner calls the subtitle detector inline for und text subs. The converter runs detection on und tracks before building the ffmpeg command (gated on a default-on `auto_detect_languages` setting). On-demand + batch API endpoints let the user trigger it manually. Results persist to the existing `audio_tracks_json` / `subtitle_tracks_json` / `native_language` / `language_source` columns; a new `has_und_tracks_flag` surfaces titles in the audio-cleanup filter.

**Tech Stack:** Python 3.11, faster-whisper (CTranslate2), langdetect, ffmpeg (already present), FastAPI, SQLite/aiosqlite, pytest.

**Reference docs:**
- Spec: `.superpowers/specs/2026-07-08-unknown-track-language-detection-design.md`
- ISO 639-1→2 map + T→B normalizer: `backend/disc_metadata.py:25` (`_ISO639_1_TO_2`), `:75` (`_normalize_iso639_2`)
- classify fns: `backend/scanner.py:404` (`detect_native_language`), `:807` (`classify_audio_tracks`), `:1050` (`classify_subtitle_tracks`)

```python
"""Language detection for tracks Shrinkerr can't tag from container/IFO/
mpls/libbluray metadata (they come through as `und`).

Two entry points:
  - detect_subtitle_language(text) — text subs, via langdetect (cheap).
  - detect_audio_language(file, stream_index) — audio, via faster-whisper
    tiny language-ID on a 30s clip (added in a later task).

Both return (ISO 639-2 code, confidence 0-1) or (None, 0.0) when detection
fails or confidence is below the configured threshold. Fail-open: callers
leave the track `und` on (None, 0.0). v0.8.0+.
"""
from __future__ import annotations

import os

from backend.disc_metadata import _ISO639_1_TO_2, _normalize_iso639_2


def _sub_min_confidence() -> float:
    try:
        return float(os.environ.get("SHRINKERR_LANG_DETECT_SUB_MIN", "0.7"))
    except ValueError:
        return 0.7


def detect_subtitle_language(text: str) -> tuple[str | None, float]:
    """Detect the language of subtitle text. Returns (ISO 639-2, confidence)
    or (None, 0.0). langdetect yields ISO 639-1 (2-letter) + a probability;
    we map to 639-2 (3-letter, B-form) via the disc_metadata tables to match
    the rest of the codebase."""
    if not text or not text.strip():
        return (None, 0.0)
    try:
        # detect_langs returns [Language(lang='en', prob=0.99), ...] sorted
        # by probability. Seed the RNG for determinism (langdetect is
        # nondeterministic by default).
        from langdetect import detect_langs, DetectorFactory
        DetectorFactory.seed = 0
        results = detect_langs(text)
    except Exception:
        return (None, 0.0)
    if not results:
        return (None, 0.0)
    best = results[0]
    iso1 = best.lang.lower()
    conf = float(best.prob)
    if conf < _sub_min_confidence():
        return (None, 0.0)
    iso2 = _ISO639_1_TO_2.get(iso1, "")
    if not iso2:
        return (None, 0.0)
    return (_normalize_iso639_2(iso2), conf)
```

NOTE: verify `_ISO639_1_TO_2` covers en→eng, de→deu, es→spa, fr→fra, ja→jpn. If a needed mapping is missing, add it to the dict in `backend/disc_metadata.py` (that's the shared table). The `_normalize_iso639_2` call converts T-form (deu) to B-form (ger) to match ffprobe/codebase convention — that's why the German test expects `ger` not `deu`.

```python
import asyncio
import tempfile

# Lazy module-global so the tiny model loads once per process (first
# audio-detection call). Downloads to the faster-whisper cache; we point
# that at the data volume so it persists across container restarts.
_WHISPER_MODEL = None
_WHISPER_LOAD_FAILED = False


def _audio_min_confidence() -> float:
    try:
        return float(os.environ.get("SHRINKERR_LANG_DETECT_AUDIO_MIN", "0.6"))
    except ValueError:
        return 0.6


def _build_audio_clip_cmd(input_path: str, stream_index: int, seek: float, out_path: str) -> list[str]:
    """ffmpeg command: 30s, mono, 16 kHz PCM WAV from `stream_index`,
    seeking to `seek` seconds. 16 kHz mono is what Whisper expects."""
    return [
        "ffmpeg", "-y", "-hide_banner", "-loglevel", "error",
        "-ss", str(seek),
        "-i", input_path,
        "-map", f"0:{stream_index}",
        "-t", "30",
        "-ac", "1",
        "-ar", "16000",
        "-c:a", "pcm_s16le",
        out_path,
    ]


async def _extract_audio_clip(input_path: str, stream_index: int, duration: float) -> str:
    """Extract a 30s clip from ~33% into the file to a temp WAV. Returns the
    temp path. Raises on ffmpeg failure (caller fail-opens)."""
    seek = max(0.0, duration * 0.33) if duration and duration > 90 else 0.0
    fd, out_path = tempfile.mkstemp(suffix=".wav", prefix="shrinkerr_lang_")
    os.close(fd)
    cmd = _build_audio_clip_cmd(input_path, stream_index, seek, out_path)
    proc = await asyncio.create_subprocess_exec(
        *cmd, stdout=asyncio.subprocess.DEVNULL, stderr=asyncio.subprocess.PIPE,
    )
    _, stderr = await asyncio.wait_for(proc.communicate(), timeout=120)
    if proc.returncode != 0:
        try:
            os.unlink(out_path)
        except OSError:
            pass
        raise OSError(f"ffmpeg clip extract failed: {stderr.decode(errors='replace')[-300:]}")
    return out_path


def _get_whisper_model():
    """Load the tiny model once. Returns None if load/download fails
    (offline first-use) — caller fail-opens."""
    global _WHISPER_MODEL, _WHISPER_LOAD_FAILED
    if _WHISPER_MODEL is not None:
        return _WHISPER_MODEL
    if _WHISPER_LOAD_FAILED:
        return None
    try:
        from faster_whisper import WhisperModel
        # Cache into the data volume so the ~75 MB download persists.
        cache_dir = os.environ.get("SHRINKERR_WHISPER_CACHE", "/app/data/models")
        os.makedirs(cache_dir, exist_ok=True)
        # device="auto" uses CUDA when available (NVENC images), else CPU.
        _WHISPER_MODEL = WhisperModel(
            "tiny", device="auto", compute_type="int8", download_root=cache_dir,
        )
        return _WHISPER_MODEL
    except Exception as exc:
        print(f"[LANG-DETECT] Whisper model load failed, audio detection disabled: {exc}", flush=True)
        _WHISPER_LOAD_FAILED = True
        return None


def _run_whisper_lang(clip_path: str) -> tuple[str | None, float]:
    """Run faster-whisper language ID on a clip. Returns (ISO 639-1, prob)
    or (None, 0.0). Isolated so tests mock this single seam."""
    model = _get_whisper_model()
    if model is None:
        return (None, 0.0)
    # faster-whisper: transcribe() returns (segments, info); info.language +
    # info.language_probability come from the LID pass over the first 30s.
    # We don't consume the segments generator (no full transcription needed).
    _segments, info = model.transcribe(clip_path, language=None)
    return (getattr(info, "language", None), float(getattr(info, "language_probability", 0.0)))


async def detect_audio_language(file_path: str, stream_index: int, duration: float = 0.0) -> tuple[str | None, float]:
    """Detect an audio track's spoken language. Extracts a 30s clip and runs
    faster-whisper language ID. Returns (ISO 639-2, confidence) or (None, 0.0).
    Fail-open on any error."""
    clip_path = None
    try:
        clip_path = await _extract_audio_clip(file_path, stream_index, duration)
        # _run_whisper_lang is sync (model inference) — run in a thread so we
        # don't block the event loop.
        iso1, conf = await asyncio.get_event_loop().run_in_executor(
            None, _run_whisper_lang, clip_path
        )
    except Exception as exc:
        print(f"[LANG-DETECT] audio detection failed for {file_path} s{stream_index}: {exc}", flush=True)
        return (None, 0.0)
    finally:
        if clip_path and os.path.exists(clip_path):
            try:
                os.unlink(clip_path)
            except OSError:
                pass
    if not iso1 or conf < _audio_min_confidence():
        return (None, 0.0)
    iso2 = _ISO639_1_TO_2.get(iso1.lower(), "")
    if not iso2:
        return (None, 0.0)
    return (_normalize_iso639_2(iso2), conf)
```

NOTE on the test mock: the test patches `_run_whisper_lang` (not the model), so the `model.transcribe` API shape doesn't need to be exercised in CI. If the real faster-whisper API differs (e.g. returns info differently across versions), that surfaces in live testing on the NUC, not in unit tests — flag this to the controller as a live-verify item.

# Text subtitle codecs whose language we can detect from their text.
# Image codecs (hdmv_pgs_subtitle, dvd_subtitle, dvb_subtitle, vobsub) are
# out of scope for v0.8.0 (OCR is v0.8.1).
_TEXT_SUB_CODECS = {"subrip", "srt", "ass", "ssa", "webvtt", "mov_text", "tx3g"}


def maybe_detect_subtitle_track_language(language: str, codec: str, text: str | None) -> str:
    """Return an upgraded ISO 639-2 language for a subtitle track, or the
    original `language` unchanged. Only detects for und TEXT subs with
    extractable text. Image subs and already-tagged tracks pass through."""
    if (language or "und").lower() != "und":
        return language
    if (codec or "").lower() not in _TEXT_SUB_CODECS:
        return language
    if not text:
        return language
    detected, _conf = detect_subtitle_language(text)
    return detected if detected else "und"
```

def _extract_embedded_sub_text(file_path: str, stream_index: int, max_chars: int = 4000) -> str | None:
    """Extract up to ~max_chars of text from an embedded text subtitle
    stream via ffmpeg (srt output to stdout). Returns None on failure."""
    import subprocess
    cmd = [
        "ffmpeg", "-v", "quiet", "-i", file_path,
        "-map", f"0:{stream_index}", "-f", "srt", "-",
    ]
    try:
        out = subprocess.run(cmd, capture_output=True, timeout=60).stdout
    except (subprocess.SubprocessError, OSError):
        return None
    if not out:
        return None
    text = out.decode(errors="replace")
    # Strip srt sequence numbers + timestamps, keep dialogue for langdetect.
    import re as _re
    text = _re.sub(r"^\d+\s*$", "", text, flags=_re.MULTILINE)
    text = _re.sub(r"\d{2}:\d{2}:\d{2},\d{3} --> .*$", "", text, flags=_re.MULTILINE)
    return text[:max_chars].strip() or None
```

@router.post("/detect-languages")
async def detect_languages(req: DetectLanguagesRequest):
    """Detect languages for a file's und audio + text-subtitle tracks,
    apply above-threshold results to scan_results, return the updated
    track lists. Fail-open per track."""
    import json as _json
    from backend.scanner import probe_file
    from backend.language_detection import (
        detect_audio_language, maybe_detect_subtitle_track_language,
    )
    probe = await probe_file(req.file_path)
    if probe is None:
        raise HTTPException(404, "Could not probe file")
    duration = probe.get("duration", 0.0) or 0.0
    changed = False
    audio = probe.get("audio_tracks", [])
    for t in audio:
        if (t.get("language") or "und").lower() == "und" and t.get("stream_index") is not None:
            lang, _c = await detect_audio_language(req.file_path, t["stream_index"], duration=duration)
            if lang:
                t["language"] = lang
                changed = True
    # (Subtitle text detection already runs at scan; re-running it here for
    # any still-und text sub is cheap — optional, mirror the scanner helper.)
    if changed:
        # Persist patched audio languages to scan_results.
        db = await connect_db()
        try:
            await db.execute(
                "UPDATE scan_results SET audio_tracks_json = ?, language_source = 'detected' "
                "WHERE file_path = ?",
                (_json.dumps(audio), req.file_path),
            )
            await db.commit()
        finally:
            await db.close()
    return {"status": "ok", "changed": changed, "audio_tracks": audio}

@router.post("/detect-languages-batch")
async def detect_languages_batch(req: DetectLanguagesBatchRequest):
    """Run detect-languages over a list of files, sequentially (single
    model instance — no parallel inference). Returns per-file results."""
    results = []
    for fp in req.file_paths:

    return {"status": "ok", "results": results}
